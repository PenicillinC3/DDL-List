# DDL 清单安卓应用 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个 Kotlin + Jetpack Compose 的安卓应用「DDL 清单」：本地管理 Deadline 条目，在自定义提前量准点发系统通知，列表分组展示并实时倒计时。

**Architecture:** 单模块 MVVM（Compose → ViewModel(StateFlow) → Repository → Room + AlarmScheduler）。精确提醒用 `AlarmManager.setExactAndAllowWhileIdle`（无权限降级 setAndAllowWhileIdle），开机重排由 `BootReceiver` 完成。纯逻辑（提醒时刻计算、分组）与平台解耦以便 JVM 单测。

**Tech Stack:** Kotlin 2.4.21 · AGP 9.4.1 · Gradle 9.6.0 · Jetpack Compose BOM 2026.09.00（Material3）· Room 2.8.5（KSP 2.3.12）· minSdk 26 / compileSdk 36 · JUnit4 + Fake 依赖单测

**Spec:** `docs/superpowers/specs/2026-10-19-ddl-app-design.md`

```kotlin
package com.gammac3.ddlapp.logic

import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test

class ReminderTimeCalculatorTest {

    private val now = 200_000_000_000L // 任意「现在」

    @Test
    fun `到期时刻本身被排入（lead 0）`() {
        val deadline = now + 10 * 60_000L
        val triggers = ReminderTimeCalculator.futureTriggers(deadline, emptyList(), now)
        assertEquals(listOf(TriggerPoint(0L, deadline)), triggers)
    }

    @Test
    fun `提前量按 deadline 减去 minute 数换算`() {
        val deadline = now + 10 * 60_000L
        val triggers = ReminderTimeCalculator.futureTriggers(deadline, listOf(5L), now)
        assertEquals(listOf(TriggerPoint(5L, deadline - 5 * 60_000L)), triggers)
    }

    @Test
    fun `过去时刻被过滤`() {
        val deadline = now + 3 * 60_000L // 3 分钟后到期
        val triggers = ReminderTimeCalculator.futureTriggers(deadline, listOf(5L, 2L), now)
        // lead=5 已在过去；lead=2 与 lead=0 仍在未来
        assertEquals(false, triggers.any { it.leadMinutes == 5L })
        assertEquals(2, triggers.size)
    }

    @Test
    fun `重复提前量去重`() {
        val deadline = now + 30 * 60_000L
        val triggers = ReminderTimeCalculator.futureTriggers(deadline, listOf(10L, 10L), now)
        assertEquals(2, triggers.size) // lead=10 + lead=0
    }

    @Test
    fun `按触发时间升序`() {
        val deadline = now + (3 * 24 * 60 + 30) * 60_000L
        val triggers = ReminderTimeCalculator.futureTriggers(deadline, listOf(1440L, 60L, 5L), now)
        assertTrue(triggers.zipWithNext().all { (a, b) -> a.triggerAtMillis < b.triggerAtMillis })
        assertEquals(0L, triggers.last().leadMinutes)
    }

    @Test
    fun `全部提前量都在过去时只剩到期时刻`() {
        val deadline = now + 60_000L
        val triggers = ReminderTimeCalculator.futureTriggers(deadline, listOf(60L), now)
        assertFalse(triggers.any { it.leadMinutes == 60L })
        assertTrue(triggers.any { it.leadMinutes == 0L })
    }
}
```

`logic\ReminderTimeCalculator.kt`:

```kotlin
package com.gammac3.ddlapp.logic

data class TriggerPoint(val leadMinutes: Long, val triggerAtMillis: Long)

object ReminderTimeCalculator {
    /** 返回仍需排期的触发点：所有提前量 + 到期时刻本身（lead=0），过滤已过去时刻，去重并升序。 */
    fun futureTriggers(deadlineMillis: Long, leadsMinutes: List<Long>, nowMillis: Long): List<TriggerPoint> =
        (leadsMinutes + 0L).distinct()
            .map { TriggerPoint(it, deadlineMillis - it * 60_000L) }
            .filter { it.triggerAtMillis > nowMillis }
            .sortedBy { it.triggerAtMillis }
}
```

```kotlin
package com.gammac3.ddlapp.logic

import com.gammac3.ddlapp.model.DdlGroup
import com.gammac3.ddlapp.model.DdlRow
import java.time.Instant
import java.time.LocalDate
import java.time.ZoneId

object DdlGrouper {
    /** 六组始终齐全；组内按截止时间升序。 */
    fun group(rows: List<DdlRow>, nowMillis: Long, zone: ZoneId): Map<DdlGroup, List<DdlRow>> {
        val out: MutableMap<DdlGroup, MutableList<DdlRow>> = DdlGroup.entries.associateWith { mutableListOf() }
        val today: LocalDate = Instant.ofEpochMilli(nowMillis).atZone(zone).toLocalDate()
        for (row in rows) {
            val target = when {
                row.isDone -> DdlGroup.DONE
                row.deadlineMillis < nowMillis -> DdlGroup.OVERDUE
                else -> {
                    val date = Instant.ofEpochMilli(row.deadlineMillis).atZone(zone).toLocalDate()
                    when (date) {
                        today -> DdlGroup.TODAY
                        today.plusDays(1) -> DdlGroup.TOMORROW
                        date < today.plusDays(8) -> DdlGroup.THIS_WEEK
                        else -> DdlGroup.LATER
                    }
                }
            }
            out.getValue(target).add(row)
        }
        return out.mapValues { (_, list) -> list.sortedBy { it.deadlineMillis } }
    }
}
```

```kotlin
package com.gammac3.ddlapp.data

import com.gammac3.ddlapp.logic.ReminderTimeCalculator
import com.gammac3.ddlapp.notify.AlarmScheduler
import kotlinx.coroutines.flow.Flow

class DdlRepository(
    private val dao: DdlDao,
    private val scheduler: AlarmScheduler,
    private val clock: () -> Long = System::currentTimeMillis,
) {
    fun observeAll(): Flow<List<DdlEntity>> = dao.observeAll()

    suspend fun upsert(entity: DdlEntity): Long {
        val id = if (entity.id == 0L) dao.insert(entity) else { dao.update(entity); entity.id }
        rescheduleFor(id)
        return id
    }

    /** 返回被删实体（供撤销还原），不存在则 null。 */
    suspend fun delete(id: Long): DdlEntity? {
        val entity = dao.byId(id) ?: return null
        scheduler.cancelAll(id)
        dao.deleteById(id)
        return entity
    }

    suspend fun setDone(id: Long, done: Boolean) {
        val entity = dao.byId(id) ?: return
        dao.update(entity.copy(isDone = done))
        if (done) scheduler.cancelAll(id) else rescheduleFor(id)
    }

    suspend fun rescheduleFor(id: Long) {
        val entity = dao.byId(id) ?: return
        scheduler.cancelAll(id)
        if (entity.isDone) return
        val triggers = ReminderTimeCalculator.futureTriggers(entity.deadlineEpochMillis, entity.leadsMinutes, clock())
        if (triggers.isNotEmpty()) scheduler.reschedule(id, entity.title, triggers)
    }

    suspend fun rescheduleAll() {
        dao.allActive().forEach { rescheduleFor(it.id) }
    }
}
```

```kotlin
package com.gammac3.ddlapp.notify

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.os.Build
import com.gammac3.ddlapp.logic.TriggerPoint

/** UI 里可选择的全部提前量（分钟）；到期时刻（0）不计入编辑选择、但计入候选。 */
object CandidateLeads {
    val ALL: List<Long> = listOf(0L, 5L, 15L, 30L, 60L, 180L, 1440L, 4320L, 10080L)
}

class AndroidAlarmScheduler(private val context: Context) : AlarmScheduler {

    private val alarmManager = context.getSystemService(Context.ALARM_SERVICE) as AlarmManager

    fun canScheduleExact(): Boolean =
        Build.VERSION.SDK_INT < Build.VERSION_CODES.S || alarmManager.canScheduleExactAlarms()

    override fun reschedule(ddlId: Long, title: String, triggers: List<TriggerPoint>) {
        cancelAll(ddlId)
        for (trigger in triggers) scheduleOne(ddlId, title, trigger)
    }

    override fun cancelAll(ddlId: Long) {
        for (lead in CandidateLeads.ALL) {
            val pi = pendingIntent(ddlId, lead, PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE) ?: continue
            pi.cancel()
        }
    }

    private fun scheduleOne(ddlId: Long, title: String, trigger: TriggerPoint) {
        val intent = Intent(context, AlarmReceiver::class.java).apply {
            putExtra(AlarmReceiver.EXTRA_ID, ddlId)
            putExtra(AlarmReceiver.EXTRA_TITLE, title)
            putExtra(AlarmReceiver.EXTRA_LEAD, trigger.leadMinutes)
        }
        val pi = PendingIntent.getBroadcast(
            context, requestCode(ddlId, trigger.leadMinutes), intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        if (canScheduleExact()) {
            alarmManager.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, trigger.triggerAtMillis, pi)
        } else {
            alarmManager.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, trigger.triggerAtMillis, pi)
        }
    }

    /** 取消搜索时不带 extras：PendingIntent 匹配按组件而非 extras，因此能命中同一 requestCode 的旧 alarm。 */
    private fun pendingIntent(ddlId: Long, lead: Long, flags: Int): PendingIntent? {
        val intent = Intent(context, AlarmReceiver::class.java)
        return PendingIntent.getBroadcast(context, requestCode(ddlId, lead), intent, flags)
    }

    private fun requestCode(ddlId: Long, lead: Long): Int = ((ddlId + 1) * 100L + lead + 1).toInt()
}
```

```kotlin
package com.gammac3.ddlapp.ui

import com.gammac3.ddlapp.notify.LeadLabels
import java.time.Instant
import java.time.ZoneId

object Format {
    private val zone: ZoneId get() = ZoneId.systemDefault()

    fun absolute(millis: Long): String {
        val z = Instant.ofEpochMilli(millis).atZone(zone)
        val h = z.hour.toString().padStart(2, '0')
        val m = z.minute.toString().padStart(2, '0')
        return "${z.monthValue}月${z.dayOfMonth}日 $h:$m"
    }

    /** 「剩 3天14:22」式倒计时 / 「已过期 2小时」，1 分钟跳变。 */
    fun remaining(deadlineMillis: Long, nowMillis: Long): String {
        val diff = deadlineMillis - nowMillis
        val past = diff < 0
        var d = Math.abs(diff)
        val days = d / 86_400_000L; d -= days * 86_400_000L
        val hours = d / 3_600_000L; d -= hours * 3_600_000L
        val minutes = d / 60_000L
        val body = when {
            days > 0 -> "${days}天${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}"
            hours > 0 -> "${hours}小时${minutes}分"
            minutes > 0 -> "${minutes}分钟"
            else -> "1分钟内"
        }
        return if (past) "已过期 $body" else "剩 $body"
    }

    fun leadText(minutes: Long): String = LeadLabels.text(minutes)
}
```

```kotlin
package com.gammac3.ddlapp.ui.list

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.gammac3.ddlapp.data.DdlRepository
import com.gammac3.ddlapp.model.DdlRow
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch

data class DdlDraft(
    val title: String,
    val note: String,
    val deadlineMillis: Long,
    val leadsMinutes: List<Long>,
)

enum class DdlFilter { ALL, ACTIVE, DONE }

data class DdlListUiState(
    val rows: List<DdlRow> = emptyList(),
    val filter: DdlFilter = DdlFilter.ALL,
    val lastDeleted: DdlRow? = null,
)

class DdlListViewModel(private val repository: DdlRepository) : ViewModel() {

    private val filter = MutableStateFlow(DdlFilter.ALL)
    private val lastDeleted = MutableStateFlow<DdlRow?>(null)

    val uiState: StateFlow<DdlListUiState> =
        combine(repository.observeAll(), filter, lastDeleted) { entities, f, deleted ->
            DdlListUiState(
                rows = entities
                    .filter { e ->
                        when (f) { DdlFilter.ALL -> true; DdlFilter.ACTIVE -> !e.isDone; DdlFilter.DONE -> e.isDone }
                    }
                    .map { e -> DdlRow(e.id, e.title, e.note, e.deadlineEpochMillis, e.leadsMinutes, e.isDone) },
                filter = f,
                lastDeleted = deleted?.let { d -> DdlRow(d.id, d.title, d.note, d.deadlineEpochMillis, d.leadsMinutes, d.isDone) },
            )
        }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), DdlListUiState())

    fun setFilter(f: DdlFilter) { filter.value = f }

    fun save(draft: DdlDraft, editingId: Long?) {
        viewModelScope.launch {
            repository.upsert(
                com.gammac3.ddlapp.data.DdlEntity(
                    id = editingId ?: 0L,
                    title = draft.title.trim(),
                    note = draft.note.trim(),
                    deadlineEpochMillis = draft.deadlineMillis,
                    leadsMinutes = draft.leadsMinutes.distinct(),
                    isDone = false,
                    createdAt = System.currentTimeMillis(),
                )
            )
        }
    }

    fun delete(row: DdlRow) {
        viewModelScope.launch {
            val snapshot = repository.delete(row.id) ?: return@launch
            lastDeleted.value = DdlRow(snapshot.id, snapshot.title, snapshot.note, snapshot.deadlineEpochMillis, snapshot.leadsMinutes, snapshot.isDone)
        }
    }

    fun restoreDeleted() {
        val d = lastDeleted.value ?: return
        viewModelScope.launch {
            repository.upsert(
                com.gammac3.ddlapp.data.DdlEntity(d.id, d.title, d.note, d.deadlineMillis, d.leadsMinutes, d.isDone, d.deadlineMillis)
            )
            lastDeleted.value = null
        }
    }

    fun toggleDone(row: DdlRow) {
        viewModelScope.launch { repository.setDone(row.id, !row.isDone) }
    }
}
```

```kotlin
@OptIn(ExperimentalMaterial3Api::class)
@Composable
private fun SwipeRow(
    row: DdlRow,
    nowMillis: Long,
    overdue: Boolean,
    onClick: () -> Unit,
    onDelete: () -> Unit,
    onToggleDone: () -> Unit,
) {
    val dismissState = rememberSwipeToDismissBoxState(
        confirmValueChange = { value ->
            if (value == SwipeToDismissBoxValue.EndToStart) {
                onDelete(); true
            } else if (value == SwipeToDismissBoxValue.StartToEnd) {
                onToggleDone(); false // 不消失，仅切换完成状态
            } else false
        }
    )
    SwipeToDismissBox(
        state = dismissState,
        enableDismissFromStartToEnd = !row.isDone,
        enableDismissFromEndToStart = true,
        backgroundContent = { /* 背景红色侧滑条 */ },
    ) {
        DdlItemCard(row = row, nowMillis = nowMillis, overdue = overdue, onClick = onClick)
    }
}
```

```kotlin
private val LEAD_CHOICES = listOf(5L, 15L, 30L, 60L, 180L, 1440L, 4320L, 10080L)

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun DdlEditorSheet(
    editing: DdlRow?,
    onDismiss: () -> Unit,
    onSave: (DdlDraft, Long?) -> Unit,
    onToggleDone: (DdlRow) -> Unit,
    onDelete: (DdlRow) -> Unit,
) {
    if (editing == null) return
    val zone = remember { ZoneId.systemDefault() }
    val nowMillis = remember { System.currentTimeMillis() }
    val initialZ: ZonedDateTime = remember(editing.id) {
        Instant.ofEpochMilli(editing.deadlineMillis).atZone(zone)
    }

    var title by remember { mutableStateOf(editing.title) }
    var note by remember { mutableStateOf(editing.note) }
    var z by remember { mutableStateOf(initialZ) }
    var leads by remember { mutableStateOf(editing.leadsMinutes.toSet()) }

    var showDate by remember { mutableStateOf(false) }
    var showTime by remember { mutableStateOf(false) }

    val canSave = title.isNotBlank() &&
        (editing.id != 0L || ZonedDateTime.now(zone).toInstant().toEpochMilli() < z.toInstant().toEpochMilli())

    val sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
    ModalBottomSheet(onDismissRequest = onDismiss, sheetState = sheetState) {
        Column(
            Modifier
                .fillMaxWidth()
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp, vertical = 8.dp),
            verticalArrangement = Arrangement.spacedBy(10.dp),
        ) {
            Text(if (editing.id == 0L) "添加 DDL" else "编辑 DDL", style = MaterialTheme.typography.titleLarge)
            OutlinedTextField(
                value = title, onValueChange = { title = it }, label = { Text("标题（必填）") },
                modifier = Modifier.fillMaxWidth(), singleLine = true,
            )
            Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                OutlinedTextField(
                    value = "📅 ${z.year}年${z.monthValue}月${z.dayOfMonth}日",
                    onValueChange = {}, readOnly = true, label = { Text("日期") },
                    modifier = Modifier.weight(3f), singleLine = true,
                    trailingIcon = { TextButton(onClick = { showDate = true }) { Text("改") } },
                )
                OutlinedTextField(
                    value = "⏰ ${z.hour.toString().padStart(2, '0')}:${z.minute.toString().padStart(2, '0')}",
                    onValueChange = {}, readOnly = true, label = { Text("时间") },
                    modifier = Modifier.weight(2f), singleLine = true,
                    trailingIcon = { TextButton(onClick = { showTime = true }) { Text("改") } },
                )
            }
            OutlinedTextField(
                value = note, onValueChange = { note = it }, label = { Text("备注（可选）") },
                modifier = Modifier.fillMaxWidth(), singleLine = true,
            )
            Text("提前提醒（可多选）", style = MaterialTheme.typography.titleSmall)
            Row(
                Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp),
            ) {
                LEAD_CHOICES.forEach { lead ->
                    val selected = lead in leads
                    FilterChip(
                        selected = selected,
                        label = { Text(LeadLabels.text(lead)) },
                        onClick = { leads = if (selected) leads - lead else leads + lead },
                    )
                }
            }
            if (editing.id == 0L && ZonedDateTime.now(zone).toInstant().toEpochMilli() >= z.toInstant().toEpochMilli()) {
                Text("截止时间必须晚于现在", color = MaterialTheme.colorScheme.error, style = MaterialTheme.typography.labelMedium)
            }
            Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                Button(
                    enabled = canSave,
                    onClick = {
                        onSave(
                            DdlDraft(
                                title = title,
                                note = note,
                                deadlineMillis = z.toInstant().toEpochMilli(),
                                leadsMinutes = leads.sorted(),
                            ),
                            editing.id.takeIf { it != 0L },
                        )
                    },
                    modifier = Modifier.weight(2f),
                ) { Text("保存") }
                if (editing.id != 0L) {
                    TextButton(onClick = { onToggleDone(editing) }, Modifier.weight(1f)) {
                        Text(if (editing.isDone) "标为未完成" else "标为已完成")
                    }
                    TextButton(onClick = { onDelete(editing) }, Modifier.weight(1f)) {
                        Text("删除", color = MaterialTheme.colorScheme.error)
                    }
                }
            }
            Spacer(Modifier.weight(1f))
        }
    }

    if (showDate) {
        val pickerState = rememberDatePickerState(
            initialSelectedDateMillis = Instant.ofEpochMilli(z.toInstant().toEpochMilli()).atZone(zone).toLocalDate()
                .atStartOfDay(zone).toInstant().toEpochMilli()
        )
        DatePickerDialog(
            onDismissRequest = { showDate = false },
            confirmButton = {
                TextButton(onClick = {
                    pickerState.selectedDateMillis?.let { picked ->
                        val newDate = Instant.ofEpochMilli(picked).atZone(zone).toLocalDate()
                        z = newDate.atTime(z.hour, z.minute).atZone(zone)
                    }
                    showDate = false
                }) { Text("确定") }
            },
            dismissButton = { TextButton(onClick = { showDate = false }) { Text("取消") } },
        ) {
            DatePicker(state = pickerState)
        }
    }
    if (showTime) {
        val timeState = rememberTimePickerState(initialHour = z.hour, initialMinute = z.minute, is24Hour = true)
        AlertDialog(
            onDismissRequest = { showTime = false },
            title = { Text("选择时间") },
            text = { TimePicker(state = timeState) },
            confirmButton = {
                TextButton(onClick = {
                    z = z.withHour(timeState.hour).withMinute(timeState.minute)
                    showTime = false
                }) { Text("确定") }
            },
            dismissButton = { TextButton(onClick = { showTime = false }) { Text("取消") } },
        )
    }
}
```
